  <script>
    async function fetchAssets() {
      const res = await fetch('/api/assets');
      const data = await res.json();
      document.getElementById('holder').textContent = data.accountHolder;
      document.getElementById('institution').textContent = data.institution;
      document.getElementById('account').textContent = data.accountNumber;
      document.getElementById('balance').textContent = `$${data.availableBalance.toFixed(2)} ${data.currency}`;
    }

    document.getElementById('pay-btn').addEventListener('click', async () => {
      const amount = document.getElementById('amount').value;
      const description = document.getElementById('desc').value;
      const statusDiv = document.getElementById('status');

      const res = await fetch('/api/pay', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ amount, description })
      });

      const data = await res.json();

      if (data.success) {
        statusDiv.className = 'status success';
        statusDiv.textContent = `${data.message} (Ref: ${data.transaction.id})`;
        fetchAssets();
      } else {
        statusDiv.className = 'status error';
        statusDiv.textContent = data.message;
      }
    });

    fetchAssets();
  </script>
